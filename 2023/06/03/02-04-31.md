dir.h-  unsigned int len;
dir.h-  char name[FLEX_ARRAY]; /* more */

--
name-hash.c:struct dir_entry {
name-hash.c-    struct hashmap_entry ent;
name-hash.c-    struct dir_entry *parent;
name-hash.c-    int nr;
name-hash.c-    unsigned int namelen;
