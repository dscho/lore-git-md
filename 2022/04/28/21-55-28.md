This is about as portable as I can find and works even in ksh. It could be optimized.

is_root() {
```
  id -u >u
  id -u root >r
  cmp -s u r
  if [ $? -ne 0 ]; then
    echo 0
  else
    echo 1
  fi
```

}

if [ `is_root` -ne 0 ]; then
        echo root
else
        echo Not root
fi
