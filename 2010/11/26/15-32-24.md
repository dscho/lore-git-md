+test_expect_success '#25: setup' '
+	unset GIT_DIR GIT_WORK_TREE &&

+	git init &&
+	git config core.bare true &&
+	GIT_WORK_TREE=non-existent &&
+	export GIT_WORK_TREE &&
+	mv .git ../25.git &&
+	echo gitdir: ../25.git >.git &&
+	cd ..
