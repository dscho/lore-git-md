#!/usr/bin/perl

use strict;
use warnings;

my ($in, $out, $line);

open $in,  '<', "Documentation/config.txt"
	or die "$!";

open $out, '>', "Documentation/config.txt+"
	or die "$!";
while ($line = <$in>) {

}

	or die "$!";

	print {$out} $line;
}

	or die "$!";
