+		if (!s)

+	}

+	if (!s) {
+		ret = error("unable to find ref alias '%s'", alias.buf);
+		goto done;

+	extract_alias_params(cmd.buf, &params);

+		param->value = xstrdup(real_ref->buf);
+	else if (param) {
+		ret = error("%%ref%% not found in alias '%s'", cmd.buf);
+		goto done;
+	}
