I think Dscho is right.  "ifdef FOO" yields (and should yield) false if
FOO is set to empty.

-- >8 -- cut here and try it for yourself -- >8 --
#!/bin/sh

rm -f config.mk

cat >Makefile <<\EOF
# The default is...
all::
FROTZ = NITFOL
-include config.mk
ifdef FROTZ
all::
	echo FROTZ is $(FROTZ)
else
all::
	echo FROTZ is not defined
endif
EOF

make
# says "NITFOL"

echo "FROTZ =" >config.mk
make
# says "not defined"

exit
