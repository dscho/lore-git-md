```
  my @opts = qw(--env deployment);
  while (<>) {
  	chomp;

  	next if /^\s*#/; # skip comments
  	if (/^\s*(.*?)\s*=\s*(.*)\s*$/) {
  		push @opts, '--'.lc($1), $2;

  }
