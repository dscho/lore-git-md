+static int norm_sha1_read(struct cache_entry **cache, unsigned int entries,

+		unsigned int entry_no = ntohl(data->entry_no);
+		if (entry_no < entries) {
+			cache[entry_no]->norm_flags = ntohl(data->norm_flags);

+	}
