+# remotes, in which case we only display the remote names

+		git_remotes_list($remotedata, $limit);

+				{-class=>"remote", -id=>$remote},
+				["remotes", $remote, $remote],
+				sub { git_remote_block($remote, $rdata, $limit, $head) }
