+check_ref_exists_1()
+{
+	symbolic=$(git rev-parse --quiet --symbolic-full-name --verify "$1")
+ 	if test "${symbolic#refs/}" != "${symbolic}"
+	then
+		echo "Reference '$1' exists."
+	else
