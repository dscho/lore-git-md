Just for giggles, I was curious who introduced the styles:

```
  pattern_authors() {
    git grep -n "$@" |
    while IFS=: read file line match; do
      git blame -L "$line,$line" "$file"
    done |
    perl -lpe '/\((.*?) \d+-\d+-\d+/; $_=$1'
  }

  $ pattern_authors ' ()' t/*.sh | sort | uniq -c | sort -rn
```
