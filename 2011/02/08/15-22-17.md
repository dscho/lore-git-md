+int parseopt_column_callback(const struct option *opt, const char *arg, int unset)
+{
+	struct column_layout *cp = opt->value;
+	if (unset) {
+		cp->mode = COL_PLAIN;
+		return 0;
+	}
+	if (!arg || !strcmp(arg, "column")) {
+		cp->mode |= COL_COLUMN_FIRST;
+		return 0;
+	}
+	if (!strcmp(arg, "dense")) {
+		cp->mode |= COL_DENSE;
+		return 0;
+	}
+	return error("unsupported style '%s'", arg);
+}
