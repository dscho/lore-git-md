+		revs->no_walk = REVISION_WALK_NO_WALK_SORTED;
+	} else {
+		/*
+		 * Detached form ("--no-walk X" as opposed to "--no-walk=X")

+		else
+			return error(_("invalid argument to --no-walk"));
+	}
+	return 0;
