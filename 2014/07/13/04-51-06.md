+void report_linked_checkout_garbage(void)
+{
+	struct strbuf sb = STRBUF_INIT;

+
+	if (!git_common_dir_env)

+		const char *path = *p;

+		strbuf_addstr(&sb, path);

+			report_garbage("unused in linked checkout", sb.buf);
+	}
+	strbuf_release(&sb);
