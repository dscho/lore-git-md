		  cd .git/hooks &&

		  cat >pre-receive <<-'EOF' &&
		  #!/bin/sh
		  cat - >>pre-receive.actual

		  cat >update <<-'EOF' &&
		  #!/bin/sh
		  printf "%s %s %s\n" "$@" >>update.actual

		  #!/bin/sh
		  cat - >>post-receive.actual

		  #!/bin/sh
		  for ref in "$@"

		  done

	  )
  }
