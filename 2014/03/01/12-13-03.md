+
+	if (!file_exists(git_path("commondir")))
+		return;
+	strbuf_addf(&sb, "%s/", get_git_dir());
+	len = sb.len;
+	for (p = list; *p; p++) {
+		strbuf_setlen(&sb, len);
+		strbuf_addstr(&sb, *p);
+		if (file_exists(sb.buf))
+			report_garbage("unused in linked checkout", sb.buf);
+	}
+	strbuf_release(&sb);
