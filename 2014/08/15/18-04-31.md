+	# - MacOS 10.0.* and MacOS 10.1.0 = Darwin 1.*

+	# i.e. "begins with [15678] and a dot" means "10.4.* or older".
 	ifeq ($(shell expr "$(uname_R)" : '[15678]\.'),2)
