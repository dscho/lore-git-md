(vaguely) indicate that this is the case.

```

# "REMOTE", "MERGED"). If not found, use "MERGE" as the default file
# where changes will be saved.

if echo "$LAYOUT" | grep @LOCAL >/dev/null
then
```

FINAL_TARGET="LOCAL"
elif echo "$LAYOUT" | grep @BASE >/dev/null
