+	free_unknown_options_usage, PARSE_OPT_KEEP_UNKNOWN_OPT);

+    printf("a = %s\n", a? "true": "false");
+    printf("b = %s\n", b? "true": "false");
+
+    int i;

+	printf("free unknown option: %s\n", unknown_opts->v[i]);
+    }
+    strvec_clear(unknown_opts);
