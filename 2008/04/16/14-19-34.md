+# Add a new submodule to the working tree, .gitmodules and the index
+#
+# $@ = repo [path]

+cmd_add()
+{
+	# parse $args after "submodule ... add".
+	while test $# -ne 0
+	do
+		case "$1" in

+			case "$2" in '') usage ;; esac

+			shift
+			;;

+			;;

+			break
+			;;
+		esac
+		shift
+	done
