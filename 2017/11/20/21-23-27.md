+	 * Separate original protocol components prog and path

+	 * will cause older git-daemon servers to crash.
+	 */
+	strbuf_addf(&request,

+		    prog, path, 0,

+

+			    get_protocol_version_config(), '\0');
+	}
+
+	packet_write(fd[1], request.buf, request.len);
+
+	free(target_host);
+	strbuf_release(&request);
+	return conn;
+}
