+test_description='Branch deletion.'
+
+. ./test-lib.sh
+
+test_expect_success \
+    'Create a new branch' \
+    '
+    stg branch --create new &&
+    git rev-parse new~0
+    '
+
+test_expect_success \
+    'Delete the new branch' \
+    'stg branch --delete new'
+
+test_expect_failure \
+    'Check the deleted branch is gone' \
+    'git rev-parse new~0'
+
+test_expect_success \
+    'Rename master branch, and create a new branch' \
+    '
+    git branch -m master a &&
+    stg branch --create b
+    '
+
+test_expect_success \
+    'Delete the new branch' \
+    'stg branch --delete b'
+
+test_expect_failure \
+    'Check the deleted branch is gone' \
+    'git rev-parse b~0'
+
+test_done
