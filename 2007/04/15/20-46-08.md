+function getCookie(name)
+{
+  var name=name+"=";
+  var c=document.cookie;
+  var p=c.indexOf(name);
+  if(p==-1)
+    return null;
+  c=c.substr(p+name.length,c.length);
+  p=c.indexOf(";");
+  if(p==-1)
+    return c;
+  else
+    return c.substr(0, p);
+}
