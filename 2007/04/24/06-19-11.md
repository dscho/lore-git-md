I second that.  :-)

$ cat config.mak
...
current_branch := \
	$(subst refs/heads/,,$(shell git-symbolic-ref HEAD 2>/dev/null))
ifeq ($(current_branch),)
	prefix := $(HOME)/sw/git-unknownbuild
else
	prefix := $(HOME)/sw/git-$(current_branch)
