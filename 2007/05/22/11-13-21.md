       if (new_whitespace == strip_whitespace && trailing_added_lines)  {

	int n = 0;

		if (newsize == n)  {
			n++;
			break;
		}
		if (new[newsize - 1 - n] != '\n')

	}
             trailing_added_lines = (n>0) ? --n : 0;
      }  else
	trailing_added_lines = 0;
