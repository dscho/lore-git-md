I stole this part from "git-mergetool.sh":

init_merge_tool_path() {
```
        merge_tool_path=`git config mergetool.$1.path`
        if test -z "$merge_tool_path" ; then
                case "$1" in
                        emerge)
                                merge_tool_path=emacs
                                ;;
                        *)
                                merge_tool_path=$1
                                ;;
                esac
        fi
```

}
