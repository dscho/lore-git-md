+struct ref *alloc_ref(unsigned namelen)
+{
+	struct ref *ret = xmalloc(sizeof(struct ref) + namelen);
+	memset(ret, 0, sizeof(struct ref) + namelen);
+	return ret;
+}
+
+void free_refs(struct ref *ref)
+{
+	struct ref *next;
+	while (ref) {
+		next = ref->next;
+		if (ref->peer_ref)
+			free(ref->peer_ref);
+		free(ref);
+		ref = next;
+	}
+}
