+        section = 'branch "%s"' % self.__name

+            return config.get(section, 'merge')
+        elif rev_parse('heads/origin'):
+            return 'heads/origin'

+            raise StackException, 'Cannot find a parent branch for "%s"' % self.__name

+    def __set_parent_branch(self, name):
+        section = 'branch "%s"' % self.__name
