#!/bin/bash

function addfile {
```
    name=$1
    echo ${name} > ${name}
    git add ${name}
    git commit -m "Add ${name}"
```

}

mkdir rebasetest1
cd rebasetest1

addfile README

git checkout -b work

git rebase -X subtree=files --onto master master
