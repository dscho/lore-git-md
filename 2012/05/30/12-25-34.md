+static int is_fmt = 0;
+static int shred(int ch)
+{
+	const char *terminators = "diouxXeEfFgGaAcsp%";
+	if (is_fmt && strchr(terminators, ch))
+		is_fmt = 0;
+	else if (!is_fmt && isprint(ch)) {
+		if (ch == '%')
+			is_fmt = 1;
+		else
+			ch = '.';
+	}
+	return ch;
+}

+			buf[gotten++] = ch;
+			if (gotten == len) {
