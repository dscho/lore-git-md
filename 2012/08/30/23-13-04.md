#!/bin/sh

    --max-parents=2 --min-parents=2 --parents v3.5..v3.6-rc2 >RL

```
	while read result parent1 parent2
	do
		$GIT merge-base $parent1 $parent2
	done <RL
```

'

GIT=../git.git/git time sh -c "$cmd" >:optim
cmp :stock :optim
