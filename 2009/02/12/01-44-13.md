struct remote {
```
	const char *name;

	const char **url;
	int url_nr;
	int url_alloc;

	struct refspec *push;
	int push_refspec_nr;

	int mirror;

	const char *uploadpack;

	/*
	 * for curl remotes only
	 */
	char *http_proxy;
```

};
