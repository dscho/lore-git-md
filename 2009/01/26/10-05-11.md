Hi Ramsay,

thanks for this idea. I solved the problem by using the bash script
included below.

Cheerz,
Steffen


> sj

```
#!/bin/sh
#

ff="/cygdrive/c/Programme/Mozilla Firefox/firefox.exe"
```


while test $# != 0
do
```
    case "$1" in
        -version)
            echo $("$ff" "-version")
            exit 0
            ;;
        -new-tab)
            p=
            case "$2" in
                http*)
                    p="$2"
                    ;;
                *)
                    # check if file exists
                    if [ -e "$2" ]
                    then
                      p="$(cygpath -w "$2")"
                    fi
                    ;;
            esac
            # check if $p has been set, otherwise exit with error
            if [ "$p" ]
            then
              $("$ff" "$p")
              exit 0
            else
              exit 1
            fi
            ;;
    esac
    shift
```

done

exit 1
