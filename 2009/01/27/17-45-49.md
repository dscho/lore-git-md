+set_fake_editor () {
+	echo "#!${SHELL_PATH-/bin_sh}" >fake-editor.sh

+	exit
+	;;
+esac
+test -z "$EXPECT_COUNT" ||

+		action=pick;;
+	esac
+done
+EOF
+

+	chmod a+x fake-editor.sh
