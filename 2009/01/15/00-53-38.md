+#!/bin/sh
+
+test_description='git merge
+
+Testing merge with the examples of Zooko.'
+
+. ./test-lib.sh
+
+#
+# A - D     \
+#   \ B - C - E
+#
+test_expect_success 'setup' '
+	cat <<EOF >file &&

+C

+	git commit -m A &&
+	git tag A &&

+G

+B
+C
+D
+E
+EOF

+C

+G
+G
+G
+A
+B
+C
+D
+E

+	git commit -m C &&
+	git tag C &&

+	cat <<EOF >file &&
+A
+B
+X
+D
+E
+EOF
+	git add file &&
+	git commit -m D &&
+	git tag D
+'
+
+test_expect_failure 'merge C' '

+A
+B
+C
+D
+E

+G
+G
+A

+X
+D
+E
+EOF
+	git merge C &&
