[core]
```

        filemode = true
        bare = false
        logallrefupdates = true
```

[remote "origin"]
```
        url = <url-that-supports-partial-clone>
        fetch = +refs/heads/*:refs/remotes/origin/*

        partialclonefilter = blob:none
```

[branch "master"]
```
        remote = origin
        merge = refs/heads/master

```
