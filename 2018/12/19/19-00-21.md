this, but try with this patch on top:

```

    index d856930b2e..f5d15895d3 100644
    --- a/perl/Git.pm
    +++ b/perl/Git.pm
    @@ -926,7 +926,13 @@ sub hash_and_insert_object {

    +       my $hash = <$in>;

    +           sub noes { die "blah" }
    +           noes();
    +       } else {
    +           chomp($hash);
    +       }
            unless (defined($hash)) {
```

    perl -d $(git --exec-path)/git-svn

  DB<1> b Git::noes

  DB<2> c
