	elif test -n "${__git_main_with_parseopt_helper[$command]}"; then

```
	case " $__git_main_with_parseopt_helper " in
	*" $command "*)
		... Yes, $command is on the list ...

	*)
		... No, $command is not on the list ...
		;;
	esac
