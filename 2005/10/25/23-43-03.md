+		fwrite(contents, sizeof(contents)-1, 1, config);
+
+		fclose(config);
+
+		if (!lstat(path, &st)) {
+			struct stat st2;
+			if (!chmod(path, st.st_mode ^ S_IXUSR) &&
+					!lstat(path, &st2) &&
+					st.st_mode != st2.st_mode)
+				unlink(path);
+			else
+				fprintf(stderr, "Ignoring file modes\n");
