directly out of diff-tree when -p is specified.

```
    GIT_INDEX_FILE=junk
    export GIT_INDEX_FILE
    rm -f junk
    date >trash ; update-cache --add trash
    t1=$(write-tree)
    date >>trash; update-cache trash
    t2=$(write-tree)    
    GIT_EXTERNAL_DIFF=echo ./diff-tree -p $t1 $t2
```
