#/bin/sh

leading=1
emptylines=0
while read line; do
```
        if ! [ "$line" ]; then
		[ "$leading" ] && continue
		emptylines=$(($emptylines + 1))
        else
		leading=0
                while [ $emptylines -gt 0 ]; do
                        echo ""
                        emptylines=$(($emptylines - 1))
                done
                echo $line
        fi
```

done
