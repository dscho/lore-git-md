+static void show_config_source(struct strbuf *buf, FILE *fp)

+	char term = '\t';

+		fn = "";
+
+	if (!end_null)

+			strbuf_addstr(buf, fn);
+	}
+
+	if (fp)
+		fputc(term, fp);
+	else
+		strbuf_addch(buf, term);
+}
