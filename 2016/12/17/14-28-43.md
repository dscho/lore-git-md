```
	test_expect_success 'ramsay-report' '
		test_config_global filter.protocol.clean cat &&

		echo "bla" >test.r &&
		git add . &&
		GIT_TRACE=1 git commit -m "test commit 2" > trace 2>&1 &&
		grep "run_command" trace
```
