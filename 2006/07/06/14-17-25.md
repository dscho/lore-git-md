> This will not work with spaces in $PATH. I'd do something like this if

This should work with shell/builtins only, no sed/cut:

```
 path=${PATH}:
 while test -n "$path"; do
   p=${path%%:*}/$1
   test ! -x "$p" || { echo "$p"; return 0; }
   path=${path#*:}
 done
 test ! -x "$1" || { echo "$1" && return 0; }
 return 1

```
