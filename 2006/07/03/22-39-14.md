+setdate () {

+	export GIT_COMMITTER_DATE GIT_AUTHOR_DATE

+	git add one &&
+	setdate &&
+	git commit -m "Initial" &&
+
+	echo uno >one &&

+	git commit -a -m "Second" &&
+
+	git checkout -b left &&
+
+	echo $datestamp >one &&
+	setdate &&

+
+	echo $datestamp >one &&
+	setdate &&
+	git commit -a -m "Common #2" &&
+
+	git branch right &&
+
+	echo $datestamp >two &&
+	setdate &&
+	git commit -a -m "Left #3" &&
+
+	echo $datestamp >two &&
+	setdate &&
+	git commit -a -m "Left #4" &&
+

+	setdate &&
