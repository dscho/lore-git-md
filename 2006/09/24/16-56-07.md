case "$1" in
--remote=*)	remote=1;;
*)		remote=0;;
esac

case "$#,$remote" in
1,0)	exec git-archive --format=tar                     "$1";;
2,0)	exec git-archive --format=tar --prefix="$2"/      "$1";;
2,1)	exec git-archive --format=tar                "$1" "$2";;
3,1)	exec git-archive --format=tar --prefix="$3"/ "$1" "$2";;
*)	usage;;
esac
