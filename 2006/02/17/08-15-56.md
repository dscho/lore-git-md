```diff
diff --git a/git-reset.sh b/git-reset.sh
index fe53fc8..195d043 100755

@@ -88,6 +88,9 @@ case "$reset_type" in

+				while (s|/[^/]*$|| && $_ ne "") {
+					rmdir($_) or last;
+				}
 			}
 		}
 	' $tmp-exists
```
