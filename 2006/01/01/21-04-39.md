This worked for me:

mkdir /dev/shm/x
cd /dev/shm/x
rsync -Ha --delete ~/kernel/git/{powerpc,linux-2.6} .
cd linux-2.6/
cat .git/HEAD  > ../start
git-pull /dev/shm/x/powerpc
cat .git/HEAD  > ../end
git-format-patch `cat ../start` `cat ../end`
