out_free"? That is where this must go.

 	char* lock_file = strdup(git_path("config.lock"));
@@ -619,10 +619,14 @@ int git_config_set_multivar(const char*

+	if (fd > 0)
+		close(fd);
 	if (config_filename)

+	}

 }
