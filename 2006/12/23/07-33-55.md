+void read_or_die(int fd, void *buf, size_t count)
+{
+	char *p = buf;
+	ssize_t loaded;
+
+	while (count > 0) {
+		loaded = xread(fd, p, count);
+		if (loaded == 0)
+			die("unexpected end of file");
+		else if (loaded < 0) {
+			if (errno == EPIPE)
+				exit(0);
+			die("read error (%s)", strerror(errno));
+		}
+		count -= loaded;
+		p += loaded;
+	}
+}
