recursivity.

+static int resolve_ancestry(const char *str, int len, unsigned char *sha1)
+{
+	int has_suffix;
+	const char *cp, *end;
+	int num = 0, ret;
+
+	/*
+	 * "name~3" is "name^^^", "name~" is "name~1", and "name^" is "name^1".
+	 */
+	has_suffix = 0;
+	for (cp = str; *cp; cp++) {
+		int ch = *cp;
+		if ('0' <= ch && ch <= '9')
+			continue;
+		if (ch == '~' || ch == '^') {
+			if (has_suffix)
+				break;
+			has_suffix = ch;
+			continue;
+		}
+		break;
+	}
+
+	if (!has_suffix)
+		return 0;
+
+	end = cp;
+	cp = str + 1;
+	if (cp == end)
+		num = 1;
+	else
+		while (cp < end)
+			num = num * 10 + *cp++ - '0';
+	len = end - str;
+	if (has_suffix == '^')
+		ret = get_parent(sha1, num);
+	else
+		ret = get_nth_ancestor(sha1, num);
