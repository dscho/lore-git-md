+static int count_distance(struct commit_list *entry)
+{
+	int nr = 0;
+
+	while (entry) {

+
+		if (commit->object.flags & (UNINTERESTING | COUNTED))
+			break;

+			nr++;
+		commit->object.flags |= COUNTED;
