in t/lib-pack.sh when GIT_TEST_DEFUALT_HASH=sha256, ...

```

  pack_trailer () {
	  test-tool $(test_oid algo) -b <"$1" >trailer.tmp &&
	  cat trailer.tmp >>"$1" &&
	  rm -f trailer.tmp
  }

```
